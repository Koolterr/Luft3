<script>
  const AIRPORTS = {
    kat: { name: "Katrineholm Kattport International Airport", terminal: "Terminal 3", gates: [1, 2, 3] },
    mal: { name: "Malmö International Airport",               terminal: "Terminal 1", gates: [2] },
    uk:  { name: "UK International Airport",                  terminal: "Terminal 1", gates: [2] }
  };

  const ROUTES = [
    ["kat", "mal"],
    ["mal", "kat"],
    ["kat", "uk"],
    ["uk", "kat"],
    ["mal", "uk"],
    ["uk", "mal"]
  ];

  // Flygtider: tisdag, torsdag, söndag, 17:00-21:00
  const ALLOWED_DAYS = [2, 4, 0];
  const START_HOUR = 17, END_HOUR = 21;
  const DAY_NAMES = ["söndag","måndag","tisdag","onsdag","torsdag","fredag","lördag"];
  const MONTHS = ["januari","februari","mars","april","maj","juni","juli","augusti","september","oktober","november","december"];

  const $ = id => document.getElementById(id);
  const dateSel = $("date"), timeSel = $("time");
  let currentRoute = 0;

  function pad(n) { return String(n).padStart(2, "0"); }
  function randomGate(airport) {
    return airport.gates[Math.floor(Math.random() * airport.gates.length)];
  }

  function buildRoutes() {
    const list = $("routeList");
    list.innerHTML = "";
    ROUTES.forEach(([f, t], i) => {
      const row = document.createElement("label");
      row.className = "route";
      row.innerHTML = `<input type="radio" name="route" value="${i}"> ${AIRPORTS[f].name} -&gt; ${AIRPORTS[t].name}`;
      row.querySelector("input").addEventListener("change", () => selectRoute(i));
      list.appendChild(row);
    });
  }

  function selectRoute(i) {
    currentRoute = i;
    const [f, t] = ROUTES[i];
    document.querySelectorAll(".route").forEach((el, idx) => {
      el.classList.toggle("selected", idx === i);
      el.querySelector("input").checked = idx === i;
    });
    $("gateInfo").innerHTML =
      `<b>Avgång:</b> ${AIRPORTS[f].terminal}<br><b>Ankomst:</b> ${AIRPORTS[t].terminal}`;
  }

  function fillDates() {
    dateSel.innerHTML = "";
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    for (let i = 0; i < 70; i++) {
      const d = new Date(today);
      d.setDate(today.getDate() + i);
      if (!ALLOWED_DAYS.includes(d.getDay())) continue;
      const o = document.createElement("option");
      o.value = `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
      const dn = DAY_NAMES[d.getDay()];
      o.textContent = `${dn.charAt(0).toUpperCase() + dn.slice(1)} ${d.getDate()} ${MONTHS[d.getMonth()]} ${d.getFullYear()}`;
      dateSel.appendChild(o);
    }
  }

  // Sista avgång 20:30 så piloten hinner flyga innan 21:00
  function fillTimes() {
    timeSel.innerHTML = "";
    for (let h = START_HOUR; h < END_HOUR; h++) {
      for (const m of [0, 30]) {
        const o = document.createElement("option");
        o.value = `${pad(h)}:${pad(m)}`;
        o.textContent = `${pad(h)}:${pad(m)}`;
        timeSel.appendChild(o);
      }
    }
  }

  buildRoutes();
  selectRoute(0);
  fillDates();
  fillTimes();

  function showError(msg) {
    const e = $("error");
    e.textContent = msg;
    e.style.display = "block";
  }

  function validate() {
    if (!$("name").value.trim()) return "Skriv in ditt Minecraft-namn.";
    if (!dateSel.value) return "Inga tillgängliga datum hittades.";
    const [y, mo, d] = dateSel.value.split("-").map(Number);
    const dt = new Date(y, mo - 1, d);
    if (!ALLOWED_DAYS.includes(dt.getDay())) return "Piloten flyger bara tisdagar, torsdagar och söndagar.";
    const [hh, mm] = timeSel.value.split(":").map(Number);
    const mins = hh * 60 + mm;
    if (mins < START_HOUR * 60 || mins > END_HOUR * 60) return "Avgångar kan bara bokas mellan 17:00 och 21:00.";
    const dep = new Date(y, mo - 1, d, hh, mm);
    if (dep < new Date()) return "Den avgångstiden har redan passerat. Välj en senare tid.";
    return null;
  }

  function ticketNumber() {
    const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
    let s = "LH-";
    for (let i = 0; i < 6; i++) s += chars[Math.floor(Math.random() * chars.length)];
    return s;
  }

  $("bookingForm").addEventListener("submit", e => {
    e.preventDefault();
    $("error").style.display = "none";
    const err = validate();
    if (err) { showError(err); return; }

    const [f, t] = ROUTES[currentRoute];
    const from = AIRPORTS[f], to = AIRPORTS[t];

    $("tName").textContent = $("name").value.trim();
    $("tDate").textContent = dateSel.options[dateSel.selectedIndex].textContent;
    $("tTime").textContent = timeSel.value;
    $("tFromAp").textContent = from.name;
    $("tFromTerm").textContent = from.terminal;
    $("tFromGate").textContent = "Gate " + randomGate(from);
    $("tToAp").textContent = to.name;
    $("tToTerm").textContent = to.terminal;
    $("tToGate").textContent = "Gate " + randomGate(to);
    $("tNumber").textContent = ticketNumber();

    $("ticketSection").style.display = "block";
    $("ticketSection").scrollIntoView({ behavior: "smooth", block: "start" });
  });

  $("newBooking").addEventListener("click", () => {
    $("ticketSection").style.display = "none";
    window.scrollTo({ top: 0, behavior: "smooth" });
  });
  $("printBtn").addEventListener("click", () => window.print());
</script>
